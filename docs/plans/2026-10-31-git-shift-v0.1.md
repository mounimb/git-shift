# git-shift v0.1 Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Ship `git-shift` v0.1.0 to PyPI: folder-based git identities, a status/doctor
explainer, and a guard that blocks commits and pushes made with the wrong identity.

**Architecture:** git is the source of truth. Each identity is a git config file;
`rules.gitconfig` (owned by git-shift) maps folders to identities with
`includeIf "gitdir/i:..."`; the global config includes `rules.gitconfig` once. A global
`core.hooksPath` points at generated POSIX `sh` hooks that call back into git-shift.
Design: `docs/plans/2026-10-31-git-shift-design.md`.

**Tech Stack:** Python 3.10+ standard library only, git 2.13+ (2.28+ for the test
suite), optional `gh`. Dev: pytest, ruff, hatchling, GitHub Actions, PyPI Trusted
Publishing.

**Step 3: Create `src/git_shift/paths.py`**

```python
"""Filesystem locations and path normalisation shared by git-shift modules."""

from __future__ import annotations

import os
from pathlib import Path


def home() -> Path:
    """Home directory as git resolves it: HOME first, then USERPROFILE (Windows)."""
    value = os.environ.get("HOME") or os.environ.get("USERPROFILE")
    return Path(value) if value else Path.home()


def _xdg_config_home() -> Path:
    xdg = os.environ.get("XDG_CONFIG_HOME")
    return Path(xdg) if xdg else home() / ".config"


def config_dir() -> Path:
    return _xdg_config_home() / "git-shift"


def identities_dir() -> Path:
    return config_dir() / "identities"


def hooks_dir() -> Path:
    return config_dir() / "hooks"


def backups_dir() -> Path:
    return config_dir() / "backups"


def rules_file() -> Path:
    return config_dir() / "rules.gitconfig"


def state_file() -> Path:
    return config_dir() / "state.gitconfig"


def xdg_git_config() -> Path:
    """git's alternative global config file."""
    return _xdg_config_home() / "git" / "config"


def expand(path: str) -> str:
    """Expand a leading ~ with the same home directory git uses."""
    if path == "~" or path.startswith(("~/", "~\\")):
        return str(home()) + path[1:]
    return path


def to_git_path(path: str | os.PathLike[str]) -> str:
    """Absolute, symlink-resolved path with forward slashes (what git config expects)."""
    resolved = os.path.realpath(expand(os.fspath(path)))
    return resolved.replace("\\", "/")


def to_folder_rule(path: str | os.PathLike[str]) -> str:
    """Folder path for an includeIf rule: normalised, always ending with one '/'."""
    git_path = to_git_path(path)
    return git_path if git_path.endswith("/") else git_path + "/"


def is_within(path: str | os.PathLike[str], folder_rule: str) -> bool:
    """True if path is the folder itself or inside it. Case-insensitive, like gitdir/i."""
    return to_folder_rule(path).casefold().startswith(folder_rule.casefold())


def depth(folder_rule: str) -> int:
    return len([part for part in folder_rule.split("/") if part])
```

```python
"""Thin wrapper around `git config`. git-shift never parses config files itself."""

from __future__ import annotations

import re
import subprocess
from dataclasses import dataclass
from pathlib import Path

from .errors import GitShiftError


@dataclass(frozen=True)
class Origin:
    kind: str  # "file", "command line", "blob", "standard input"
    path: str  # config file path when kind == "file", else ""
    value: str


def run_git(
    *args: str, cwd: str | Path | None = None, check: bool = True
) -> subprocess.CompletedProcess[str]:
    try:
        proc = subprocess.run(
            ["git", *args],
            cwd=cwd,
            capture_output=True,
            text=True,
            encoding="utf-8",
            errors="replace",
        )
    except FileNotFoundError as exc:
        raise GitShiftError("git is not installed or not on PATH") from exc
    if check and proc.returncode != 0:
        raise GitShiftError(f"`git {' '.join(args)}` failed: {proc.stderr.strip()}")
    return proc


def _where(scope: str | None, file: str | Path | None) -> list[str]:
    if file is not None:
        return ["--file", str(file)]
    if scope is not None:
        return [f"--{scope}"]
    return []


def get(key: str, *, scope=None, file=None, cwd=None) -> str | None:
    proc = run_git("config", *_where(scope, file), "-z", "--get", key, cwd=cwd, check=False)
    return proc.stdout.split("\0")[0] if proc.returncode == 0 else None


def get_all(key: str, *, scope=None, file=None, cwd=None) -> list[str]:
    proc = run_git("config", *_where(scope, file), "-z", "--get-all", key, cwd=cwd, check=False)
    return proc.stdout.split("\0")[:-1] if proc.returncode == 0 else []


def get_with_origin(key: str, *, cwd=None) -> Origin | None:
    """Effective value of `key` in `cwd` and the place it comes from."""
    proc = run_git("config", "--show-origin", "-z", "--get", key, cwd=cwd, check=False)
    if proc.returncode != 0:
        return None
    origin, value = proc.stdout.split("\0")[:2]
    kind, _, path = origin.partition(":")
    return Origin(kind=kind, path=path, value=value)


def list_entries(*, scope=None, file=None, cwd=None) -> list[tuple[str, str]]:
    """All (key, value) pairs in file order. Keys come back lowercased except subsections."""
    proc = run_git("config", *_where(scope, file), "-z", "--list", cwd=cwd, check=False)
    if proc.returncode != 0:
        return []
    entries = []
    for record in proc.stdout.split("\0"):
        if record:
            key, sep, value = record.partition("\n")
            entries.append((key, value if sep else "true"))
    return entries


def set_value(key: str, value: str, *, scope=None, file=None, cwd=None) -> None:
    if file is not None:
        Path(file).parent.mkdir(parents=True, exist_ok=True)
    run_git("config", *_where(scope, file), key, value, cwd=cwd)


def add_value(key: str, value: str, *, scope=None, file=None) -> None:
    run_git("config", *_where(scope, file), "--add", key, value)


def unset(key: str, *, scope=None, file=None, cwd=None) -> None:
    if file is not None and not Path(file).exists():
        return
    proc = run_git("config", *_where(scope, file), "--unset-all", key, cwd=cwd, check=False)
    if proc.returncode not in (0, 5):  # 5 = key was not set
        raise GitShiftError(f"could not unset {key}: {proc.stderr.strip()}")


def git_version() -> tuple[int, int]:
    out = run_git("--version").stdout
    match = re.search(r"(\d+)\.(\d+)", out)
    if match is None:
        raise GitShiftError(f"cannot parse git version from {out!r}")
    return int(match.group(1)), int(match.group(2))

"""Parse git remote URLs into host / owner / repository."""

from __future__ import annotations

import re
from dataclasses import dataclass

_URL = re.compile(r"^(?:https?|ssh|git)://(?:[^@/]+@)?(?P<host>[^:/]+)(?::\d+)?/(?P<path>.+)$")
_SCP = re.compile(r"^(?:[^@/]+@)?(?P<host>[^:/]+):(?P<path>[^/].*)$")


@dataclass(frozen=True)
class Remote:
    host: str
    owner: str
    repo: str


def parse_remote(url: str) -> Remote | None:
    url = url.strip()
    match = _URL.match(url) or _SCP.match(url)
    if match is None:
        return None
    parts = [part for part in match.group("path").split("/") if part]
    if len(parts) < 2:
        return None
    repo = parts[-1].removesuffix(".git")
    return Remote(match.group("host").lower(), parts[0], repo)
```

```python
"""Identities: one git config file per identity, plus the rules that activate them."""

from __future__ import annotations

import re
import shlex
from dataclasses import dataclass
from pathlib import Path

from . import gitconfig, paths
from .errors import GitShiftError

NAME_RE = re.compile(r"^[a-z0-9][a-z0-9_-]*$")
EMAIL_RE = re.compile(r"^[^@\s]+@[^@\s]+\.[^@\s]+$")


@dataclass(frozen=True)
class Identity:
    name: str
    email: str
    user_name: str
    folder: str  # folder rule: absolute, forward slashes, trailing "/"
    ssh_key: str | None = None
    gh: str | None = None

    @property
    def path(self) -> Path:
        return identity_path(self.name)


def identity_path(name: str) -> Path:
    return paths.identities_dir() / f"{name}.gitconfig"


def validate_name(name: str) -> None:
    if not NAME_RE.match(name):
        raise GitShiftError(
            f'invalid identity name "{name}": use lowercase letters, digits, "-" and "_"'
        )


def validate_email(email: str) -> None:
    if not EMAIL_RE.match(email):
        raise GitShiftError(f'invalid email "{email}"')


def ssh_command(key_path: str) -> str:
    return f"ssh -i {shlex.quote(key_path)} -o IdentitiesOnly=yes"


def save(identity: Identity) -> None:
    path = identity.path
    if path.exists():
        path.unlink()
    entries = [("user.name", identity.user_name), ("user.email", identity.email)]
    if identity.ssh_key:
        entries.append(("core.sshCommand", ssh_command(identity.ssh_key)))
    entries.append(("gitshift.folder", identity.folder))
    if identity.ssh_key:
        entries.append(("gitshift.sshKey", identity.ssh_key))
    if identity.gh:
        entries.append(("gitshift.gh", identity.gh))
    for key, value in entries:
        gitconfig.set_value(key, value, file=path)


def load(name: str) -> Identity:
    path = identity_path(name)
    if not path.is_file():
        raise GitShiftError(f'no identity named "{name}"')
    values = dict(gitconfig.list_entries(file=path))  # one git call per identity
    return Identity(
        name=name,
        email=values.get("user.email", ""),
        user_name=values.get("user.name", ""),
        folder=values.get("gitshift.folder", ""),
        ssh_key=values.get("gitshift.sshkey"),
        gh=values.get("gitshift.gh"),
    )


def list_identities() -> list[Identity]:
    directory = paths.identities_dir()
    if not directory.is_dir():
        return []
    return [load(path.stem) for path in sorted(directory.glob("*.gitconfig"))]
```

```python
def write_rules(identities: list[Identity]) -> None:
    """Regenerate rules.gitconfig. Deeper folders come last so they win."""
    rules = paths.rules_file()
    rules.parent.mkdir(parents=True, exist_ok=True)
    tmp = rules.with_name(rules.name + ".tmp")
    tmp.write_text("# Managed by git-shift. Do not edit by hand.\n", encoding="utf-8")
    for identity in sorted(identities, key=lambda i: (paths.depth(i.folder), i.name)):
        gitconfig.set_value(
            f"includeIf.gitdir/i:{paths.gitdir_pattern(identity.folder)}.path",
            paths.to_git_path(identity.path),
            file=tmp,
        )
    os.replace(tmp, rules)


def global_config_file() -> Path | None:
    for candidate in paths.global_config_files():
        if candidate.is_file():
            return candidate
    return None


def backup_global_config() -> Path | None:
    """Copy the global config once, before git-shift changes it for the first time."""
    source = global_config_file()
    backups = paths.backups_dir()
    if source is None or (backups.is_dir() and any(backups.iterdir())):
        return None
    backups.mkdir(parents=True, exist_ok=True)
    target = backups / f"gitconfig-{time.strftime('%Y%m%d-%H%M%S')}"
    shutil.copy2(source, target)
    return target


def ensure_global_include() -> None:
    target = paths.to_git_path(paths.rules_file())
    current = gitconfig.get_all("include.path", scope="global")
    if target.casefold() in (paths.to_git_path(value).casefold() for value in current):
        return
    backup_global_config()
    gitconfig.add_value("include.path", target, scope="global")


IDENTITY_KEYS = ("user.name", "user.email", "core.sshcommand")


def keys_after_global_include() -> list[str] | None:
    """Identity keys the global config sets after the git-shift include (they override
    every identity), or None when the include is missing. Includes are not followed."""
    rules = paths.to_git_path(paths.rules_file()).casefold()
    entries = gitconfig.list_entries(scope="global")
    positions = [
        index
        for index, (key, value) in enumerate(entries)
        if key == "include.path" and paths.to_git_path(value).casefold() == rules
    ]
    if not positions:
        return None
    return sorted({key for key, _ in entries[positions[-1] + 1 :] if key in IDENTITY_KEYS})


def add_identity(
    name: str,
    *,
    email: str,
    user_name: str,
    folder: str,
    ssh_key: str | None = None,
    gh: str | None = None,
) -> tuple[Identity, list[str]]:
    validate_name(name)
    validate_email(email)
    if identity_path(name).exists():
        raise GitShiftError(f'identity "{name}" already exists')
    folder_rule = paths.to_folder_rule(folder)
    existing = list_identities()
    for other in existing:
        if other.folder.casefold() == folder_rule.casefold():
            raise GitShiftError(f'folder {folder_rule} is already used by "{other.name}"')

    warnings = []
    if not Path(folder_rule).is_dir():
        warnings.append(f"folder {folder_rule} does not exist yet")
    key = paths.to_git_path(ssh_key) if ssh_key else None
    if key and not Path(key).is_file():
        warnings.append(f"SSH key {key} not found")

    identity = Identity(name, email, user_name, folder_rule, key, gh)
    save(identity)
    write_rules([*existing, identity])
    ensure_global_include()
    late = keys_after_global_include()
    if late:
        warnings.append(
            f"your global config sets {', '.join(late)} after the git-shift include, "
            "so it overrides every identity: run git shift doctor"
        )
    return identity, warnings


def remove_identity(name: str) -> Identity:
    identity = load(name)
    identity.path.unlink()
    write_rules(list_identities())
    return identity
```

"""Command-line interface for git-shift."""

from __future__ import annotations

import argparse
import sys
from collections.abc import Callable

from . import __version__, store
from .errors import GitShiftError

OK, WARN, ERR = "✓", "!", "✗"


def cmd_add(args: argparse.Namespace) -> int:
    identity, warnings = store.add_identity(
        args.name,
        email=args.email,
        user_name=args.user_name,
        folder=args.folder,
        ssh_key=args.ssh_key,
        gh=args.gh,
    )
    print(f'{OK} identity "{identity.name}" created')
    print(f"  repos under {identity.folder} now commit as {identity.email}")
    for warning in warnings:
        print(f"{WARN} {warning}")
    return 0


def cmd_list(args: argparse.Namespace) -> int:
    identities = store.list_identities()
    if not identities:
        print("no identities yet. Create one with:")
        print("  git shift add <name> --email <email> --name <name> --folder <dir>")
        return 0
    width = max(len(identity.name) for identity in identities)
    for identity in identities:
        gh_part = f"  gh: {identity.gh}" if identity.gh else ""
        print(f"{identity.name:<{width}}  {identity.email}  {identity.folder}{gh_part}")
    return 0


def cmd_remove(args: argparse.Namespace) -> int:
    identity = store.remove_identity(args.name)
    print(f'{OK} identity "{identity.name}" removed')
    return 0


COMMANDS: dict[str, Callable[[argparse.Namespace], int]] = {
    "add": cmd_add,
    "list": cmd_list,
    "remove": cmd_remove,
}


def build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(
        prog="git-shift",
        description="Folder-based git identities, with a guard against wrong-account commits.",
    )
    parser.add_argument("--version", action="version", version=f"git-shift {__version__}")
    parser.add_argument("--debug", action="store_true", help="show tracebacks on errors")
    sub = parser.add_subparsers(dest="command", required=True, metavar="<command>")

    add = sub.add_parser("add", help="create an identity bound to a folder")
    add.add_argument("name", help="identity name, e.g. perso")
    add.add_argument("--email", required=True, help="commit email")
    add.add_argument("--name", dest="user_name", required=True, metavar="NAME", help="author name")
    add.add_argument("--folder", required=True, help="repos under this folder use the identity")
    add.add_argument("--ssh-key", help="private key used for git over SSH")
    add.add_argument("--gh", metavar="LOGIN", help="GitHub login expected for this identity")

    sub.add_parser("list", help="list identities")

    remove = sub.add_parser("remove", help="delete an identity")
    remove.add_argument("name")
    return parser


def _configure_output() -> None:
    """Never crash on a console that cannot print symbols like ✓."""
    for stream in (sys.stdout, sys.stderr):
        reconfigure = getattr(stream, "reconfigure", None)
        if reconfigure is not None:
            reconfigure(errors="replace")


def main(argv: list[str] | None = None) -> int:
    _configure_output()
    argv = sys.argv[1:] if argv is None else argv
    args = build_parser().parse_args(argv)
    try:
        return COMMANDS[args.command](args)
    except GitShiftError as exc:
        print(f"git-shift: error: {exc}", file=sys.stderr)
        return exc.exit_code
    except Exception as exc:
        if args.debug:
            raise
        print(f"git-shift: unexpected error: {exc} (rerun with --debug)", file=sys.stderr)
        return 1

"""Work out which identity should apply in a directory, and which one actually does."""

from __future__ import annotations

import os
from dataclasses import dataclass
from pathlib import Path

from . import gitconfig, paths, remotes
from .store import Identity


@dataclass(frozen=True)
class Effective:
    email: str | None
    name: str | None
    ssh_command: str | None
    source: str  # identity | local | global | environment | command line | other | unset
    identity: str | None = None
    origin: str | None = None


def expected_identity(path, identities: list[Identity]) -> Identity | None:
    matches = [identity for identity in identities if paths.is_within(path, identity.folder)]
    return max(matches, key=lambda identity: paths.depth(identity.folder), default=None)


def remote_identity(url: str, identities: list[Identity]) -> Identity | None:
    remote = remotes.parse_remote(url)
    if remote is None or remote.host != "github.com":
        return None
    for identity in identities:
        if identity.gh and identity.gh.casefold() == remote.owner.casefold():
            return identity
    return None


def repo_root(cwd) -> str | None:
    proc = gitconfig.run_git("rev-parse", "--show-toplevel", cwd=cwd, check=False)
    if proc.returncode != 0:
        return None
    return proc.stdout.strip() or None


def git_dir(cwd) -> str | None:
    proc = gitconfig.run_git("rev-parse", "--absolute-git-dir", cwd=cwd, check=False)
    if proc.returncode != 0:
        return None
    return proc.stdout.strip() or None


def origin_url(cwd) -> str | None:
    # get-url expands insteadOf/pushInsteadOf and follows pushurl, like the URL git hands
    # to pre-push, so commit-time and push-time checks agree (Task 6 review).
    proc = gitconfig.run_git("remote", "get-url", "--push", "origin", cwd=cwd, check=False)
    return (proc.stdout.strip() or None) if proc.returncode == 0 else None


def expected_for_repo(
    cwd, identities: list[Identity], remote_url: str | None = None
) -> tuple[Identity, str] | None:
    """The identity a repository must use, and the reason (folder first, then remote).

    Folders are matched against the git directory, like includeIf gitdir, so worktrees
    and separate git dirs agree with what git applies.
    """
    location = git_dir(cwd) or str(cwd)
    identity = expected_identity(location, identities)
    if identity is not None:
        return identity, f"folder {identity.folder}"
    url = remote_url or origin_url(cwd)
    identity = remote_identity(url, identities) if url else None
    if identity is not None:
        return identity, f"GitHub owner {identity.gh}"
    return None


def _local_config(cwd) -> str | None:
    proc = gitconfig.run_git("rev-parse", "--git-common-dir", cwd=cwd, check=False)
    if proc.returncode != 0:
        return None
    return paths.to_git_path(Path(cwd) / proc.stdout.strip() / "config")


def _classify(origin: gitconfig.Origin, identities: list[Identity], cwd) -> tuple[str, str | None]:
    if origin.kind != "file":
        return origin.kind, None
    path = paths.to_git_path(Path(cwd) / origin.path).casefold()
    for identity in identities:
        if paths.to_git_path(identity.path).casefold() == path:
            return "identity", identity.name
    if path == (_local_config(cwd) or "").casefold():
        return "local", None
    global_files = paths.global_config_files()
    if path in {paths.to_git_path(file).casefold() for file in global_files}:
        return "global", None
    return "other", None


def effective(cwd, identities: list[Identity]) -> Effective:
    name = gitconfig.get("user.name", cwd=cwd)
    ssh = gitconfig.get("core.sshCommand", cwd=cwd)
    env_email = os.environ.get("GIT_AUTHOR_EMAIL")
    if env_email:
        return Effective(env_email, name, ssh, "environment")
    origin = gitconfig.get_with_origin("user.email", cwd=cwd)
    if origin is None:
        return Effective(None, name, ssh, "unset")
    source, identity = _classify(origin, identities, cwd)
    return Effective(origin.value, name, ssh, source, identity, origin.path)
```

"""Read-only view of the GitHub CLI's logged-in accounts."""

from __future__ import annotations

import dataclasses
import os
import re
import subprocess
from dataclasses import dataclass

from . import paths

_LOGGED_IN = re.compile(r"Logged in to (\S+) (?:account|as) (\S+)")
_FAILED = re.compile(r"Failed to log in to (\S+) account (\S+)")
_ACTIVE = re.compile(r"Active account: (true|false)")


@dataclass(frozen=True)
class GhAccount:
    host: str
    login: str
    active: bool
    valid: bool


def parse_auth_status(text: str) -> list[GhAccount]:
    accounts: list[GhAccount] = []
    saw_active_flag = False
    for line in text.splitlines():
        if match := _LOGGED_IN.search(line):
            accounts.append(GhAccount(match.group(1), match.group(2), active=False, valid=True))
        elif match := _FAILED.search(line):
            accounts.append(GhAccount(match.group(1), match.group(2), active=False, valid=False))
        elif (match := _ACTIVE.search(line)) and accounts:
            saw_active_flag = True
            accounts[-1] = dataclasses.replace(accounts[-1], active=match.group(1) == "true")
    if not saw_active_flag:  # gh < 2.40: one account per host, always the active one
        accounts = [dataclasses.replace(account, active=account.valid) for account in accounts]
    return accounts


def _executable() -> str | None:
    return paths.find_executable("gh")  # never from the current directory (Task 5 review)


def accounts() -> list[GhAccount] | None:
    """Accounts known to gh, or None when gh is not installed or does not answer."""
    exe = _executable()
    if exe is None:
        return None
    env = {**os.environ, "GH_NO_UPDATE_NOTIFIER": "1", "NO_COLOR": "1", "GH_PROMPT_DISABLED": "1"}
    try:
        proc = subprocess.run(
            [exe, "auth", "status"],
            capture_output=True,
            text=True,
            encoding="utf-8",
            errors="replace",
            env=env,
            timeout=20,
        )
    except (OSError, subprocess.TimeoutExpired):
        return None
    return parse_auth_status(proc.stdout + "\n" + proc.stderr)


def active_login(accounts: list[GhAccount], host: str = "github.com") -> str | None:
    for account in accounts:
        if account.host == host and account.active and account.valid:
            return account.login
    return None
```

from __future__ import annotations

import os
import re
import shlex
import shutil
import sys
from pathlib import Path

from . import gitconfig, paths, store
from .errors import GitShiftError

GUARDED = ("pre-commit", "pre-push")
# Chain-only hooks: a global core.hooksPath disables every local hook type, so these
# keep repositories' own hooks working. reference-transaction and post-index-change are
# left out on purpose: they run on almost every git command.
PASSTHROUGH = (
    "applypatch-msg",
    "pre-applypatch",
    "post-applypatch",
    "pre-merge-commit",
    "prepare-commit-msg",
    "commit-msg",
    "post-commit",
    "pre-rebase",
    "post-checkout",
    "post-merge",
    "pre-auto-gc",
    "post-rewrite",
    "sendemail-validate",
    "push-to-checkout",
)
BLOCK_EXIT = 3  # distinct from Python's own exit codes (1 = crash, 2 = usage)
LAUNCHER = "_git_shift_launch.py"
STATE_PREVIOUS = "gitshift.previousHooksPath"
_PY_LINE = re.compile(r"^GIT_SHIFT_PY=(.+)$", re.MULTILINE)
_FEED = '{ [ -z "$GIT_SHIFT_STDIN" ] || printf \'%s\\n\' "$GIT_SHIFT_STDIN"; } | '


def render_launcher() -> str:
    package_parent = str(Path(__file__).resolve().parent.parent)
    return (
        "# git-shift hook launcher. Run with python -I so repository files are never imported.\n"
        "import sys\n"
        f"sys.path.insert(0, {package_parent!r})\n"
        "from git_shift.cli import main\n"
        "sys.exit(main(sys.argv[1:]))\n"
    )


def render_hook(hook: str, python: str, launcher: str, chain_dir: str | None) -> str:
    reads_stdin = hook == "pre-push"  # stdin must reach both the check and the chained hook
    feed = _FEED if reads_stdin else ""
    lines = ["#!/bin/sh", f"# git-shift guard ({hook}). Remove with: git shift guard uninstall"]
    if reads_stdin:
        lines.append("GIT_SHIFT_STDIN=$(cat)")
    if hook in GUARDED:
        lines += [
            f"GIT_SHIFT_PY={shlex.quote(python)}",
            f"GIT_SHIFT_LAUNCH={shlex.quote(launcher)}",
            'if [ -z "$GIT_SHIFT_SKIP" ]; then',
            '  if [ -f "$GIT_SHIFT_PY" ] && [ -f "$GIT_SHIFT_LAUNCH" ]; then',
            f'    {feed}"$GIT_SHIFT_PY" -I "$GIT_SHIFT_LAUNCH" __hook {hook} "$@"',
            f"    if [ $? -eq {BLOCK_EXIT} ]; then exit 1; fi",
            "  else",
            '    echo "git-shift: warning: $GIT_SHIFT_PY not found, guard skipped" >&2',
            "  fi",
            "fi",
        ]
    if chain_dir is None:
        lines.append('GIT_SHIFT_CHAIN="$(git rev-parse --git-common-dir)/hooks"')
    else:
        lines.append(f"GIT_SHIFT_CHAIN={shlex.quote(chain_dir)}")
    lines.append(f'if [ -x "$GIT_SHIFT_CHAIN/{hook}" ]; then')
    if reads_stdin:
        lines += [f'  {feed}"$GIT_SHIFT_CHAIN/{hook}" "$@"', "  exit $?"]
    else:
        lines.append(f'  exec "$GIT_SHIFT_CHAIN/{hook}" "$@"')
    lines += ["fi", "exit 0", ""]
    return "\n".join(lines)


def is_installed() -> bool:
    current = gitconfig.get("core.hooksPath", scope="global")
    ours = paths.to_git_path(paths.hooks_dir())
    return bool(current) and paths.to_git_path(current).casefold() == ours.casefold()


def _chain_dir(previous: str | None) -> str | None:
    if not previous:
        return None
    if previous.startswith("~") or os.path.isabs(previous):
        return paths.to_git_path(previous)
    return previous  # relative: git resolves it from each repository's top level


def install() -> str | None:
    """Install the hooks. Returns the previous global core.hooksPath that is chained."""
    state = paths.state_file()
    if is_installed():
        previous = gitconfig.get(STATE_PREVIOUS, file=state)
    else:
        previous = gitconfig.get("core.hooksPath", scope="global")
        if previous:
            gitconfig.set_value(STATE_PREVIOUS, previous, file=state)
        else:
            gitconfig.unset(STATE_PREVIOUS, file=state)

    hooks = paths.hooks_dir()
    hooks.mkdir(parents=True, exist_ok=True)
    launcher = hooks / LAUNCHER
    launcher.write_text(render_launcher(), encoding="utf-8", newline="\n")
    python = sys.executable.replace("\\", "/")
    chain_dir = _chain_dir(previous)
    for hook in GUARDED + PASSTHROUGH:
        script = hooks / hook
        text = render_hook(hook, python, paths.to_git_path(launcher), chain_dir)
        script.write_text(text, encoding="utf-8", newline="\n")
        script.chmod(0o755)

    store.backup_global_config()
    gitconfig.set_value("core.hooksPath", paths.to_git_path(hooks), scope="global")
    return previous


def uninstall() -> str | None:
    """Remove the hooks and restore the previous global core.hooksPath, if any."""
    if not is_installed():
        raise GitShiftError(
            "the guard is not installed (global core.hooksPath does not point to git-shift)"
        )
    state = paths.state_file()
    previous = gitconfig.get(STATE_PREVIOUS, file=state)
    if previous:
        gitconfig.set_value("core.hooksPath", previous, scope="global")
    else:
        gitconfig.unset("core.hooksPath", scope="global")
    gitconfig.unset(STATE_PREVIOUS, file=state)
    shutil.rmtree(paths.hooks_dir(), ignore_errors=True)
    return previous


def installed_python() -> str | None:
    script = paths.hooks_dir() / "pre-commit"
    if not script.is_file():
        return None
    match = _PY_LINE.search(script.read_text(encoding="utf-8"))
    return shlex.split(match.group(1))[0] if match else None

_ZERO_SHA = re.compile(r"^0+$")


def pushed_commits(cwd, stdin_text: str) -> dict[str, str]:
    """Commits about to be pushed, as {short sha: author email}."""
    commits: dict[str, str] = {}
    for line in stdin_text.splitlines():
        parts = line.split()
        if len(parts) != 4:
            continue
        _local_ref, local_sha, _remote_ref, remote_sha = parts
        if _ZERO_SHA.match(local_sha):  # branch deletion
            continue
        ranges = [[local_sha, "--not", "--remotes"]]
        if not _ZERO_SHA.match(remote_sha):
            ranges.insert(0, [f"{remote_sha}..{local_sha}"])
        for revisions in ranges:  # fall back when remote_sha is unknown locally
            proc = gitconfig.run_git("log", "--format=%h %ae", *revisions, cwd=cwd, check=False)
            if proc.returncode == 0:
                for row in proc.stdout.splitlines():
                    sha, _, email = row.partition(" ")
                    commits[sha] = email
                break
    return commits


def check_pre_push(cwd, args: list[str], stdin_text: str) -> int:
    url = args[1] if len(args) > 1 else None
    found = resolve.expected_for_repo(cwd, store.list_identities(), remote_url=url)
    if found is None:
        return 0
    identity, reason = found
    wrong = {
        sha: email
        for sha, email in pushed_commits(cwd, stdin_text).items()
        if email.casefold() != identity.email.casefold()
    }
    if not wrong:
        return 0
    err = sys.stderr
    print(
        f"✗ git-shift: {len(wrong)} commit(s) to push are not authored by "
        f'"{identity.name}" <{identity.email}> (matched by {reason})',
        file=err,
    )
    for sha, email in list(wrong.items())[:10]:
        print(f"  {sha}  {email}", file=err)
    print("  fix:    re-author them, e.g. git commit --amend --reset-author --no-edit", file=err)
    print("  bypass: git push --no-verify", file=err)

"""Health checks behind `git shift doctor`."""

from __future__ import annotations

from dataclasses import dataclass
from pathlib import Path

from . import gh, gitconfig, guard, paths, resolve, store


@dataclass(frozen=True)
class Finding:
    level: str  # "ok" | "warn" | "error"
    message: str
    fix: str | None = None


def run(cwd) -> list[Finding]:
    major, minor = gitconfig.git_version()
    if (major, minor) < (2, 13):
        return [Finding("error", f"git {major}.{minor} is too old: git-shift needs 2.13+", "update git")]
    findings = [Finding("ok", f"git {major}.{minor}")]
    identities = store.list_identities()
    if identities:
        findings += _check_include()
        findings += _check_identities(identities)
    else:
        findings.append(
            Finding("warn", "no identities yet", "git shift add <name> --email E --name N --folder DIR")
        )
    findings += _check_gh(identities)
    findings += _check_guard()
    findings += _check_repo(cwd, identities)
    return findings


def _check_include() -> list[Finding]:
    late = store.keys_after_global_include()
    if late is None:
        rules = paths.to_git_path(paths.rules_file())
        return [
            Finding(
                "error",
                "git-shift rules are not included from your global git config",
                f"git config --global --add include.path {rules}",
            )
        ]
    if late:
        return [
            Finding(
                "error",
                f"your global config sets {', '.join(late)} after the git-shift include, "
                "so it overrides every identity",
                "move the [include] block pointing to git-shift to the end of your global config",
            )
        ]
    return [Finding("ok", "git-shift rules are active in your global git config")]


def _check_identities(identities: list[store.Identity]) -> list[Finding]:
    findings = []
    for identity in identities:
        if not Path(identity.folder).is_dir():
            findings.append(
                Finding("warn", f'folder of "{identity.name}" does not exist: {identity.folder}')
            )
        if identity.ssh_key and not Path(identity.ssh_key).is_file():
            findings.append(
                Finding(
                    "error",
                    f'SSH key of "{identity.name}" not found: {identity.ssh_key}',
                    f"ssh-keygen -t ed25519 -f {identity.ssh_key}",
                )
            )
    if not findings:
        count = len(identities)
        findings.append(Finding("ok", f"{count} identit{'y' if count == 1 else 'ies'} configured"))
    return findings


def _check_gh(identities: list[store.Identity]) -> list[Finding]:
    accounts = gh.accounts()
    if accounts is None:
        return [Finding("ok", "gh not installed: GitHub CLI checks skipped")]
    findings = []
    for account in accounts:
        if not account.valid:
            findings.append(
                Finding(
                    "warn",
                    f"gh login for {account.login} on {account.host} is invalid",
                    f"gh auth login -h {account.host}   "
                    f"(or forget it: gh auth logout -h {account.host} -u {account.login})",
                )
            )
    valid = {a.login.casefold() for a in accounts if a.valid and a.host == "github.com"}
    for identity in identities:
        if identity.gh and identity.gh.casefold() not in valid:
            findings.append(
                Finding(
                    "error",
                    f'identity "{identity.name}" expects gh account "{identity.gh}", '
                    "which is not logged in or has an invalid token",
                    "gh auth login -h github.com",
                )
            )
    return findings or [Finding("ok", "gh logins valid")]


def _check_guard() -> list[Finding]:
    if not guard.is_installed():
        return [
            Finding(
                "warn",
                "guard not installed: wrong-identity commits are not blocked",
                "git shift guard install",
            )
        ]
    python = guard.installed_python()
    if python and not Path(python).is_file():
        return [
            Finding("error", f"guard hooks use a missing Python: {python}", "git shift guard install")
        ]
    return [Finding("ok", "guard installed")]


def _check_repo(cwd, identities: list[store.Identity]) -> list[Finding]:
    root = resolve.repo_root(cwd)
    if root is None:
        return []
    findings = []
    expected = resolve.expected_identity(root, identities)
    if expected is not None:
        eff = resolve.effective(cwd, identities)
        if eff.source == "local" and (eff.email or "").casefold() != expected.email.casefold():
            findings.append(
                Finding(
                    "warn",
                    f"this repo's local config sets user.email={eff.email}, "
                    f'overriding "{expected.name}"',
                    "git config --unset user.email",
                )
            )
    if guard.is_installed() and gitconfig.get("core.hooksPath", scope="local", cwd=cwd):
        findings.append(
            Finding(
                "warn",
                "this repo sets its own core.hooksPath (husky, pre-commit...): guard inactive here",
            )
        )
    return findings

# git-shift

[![CI](https://github.com/mounimb/git-shift/actions/workflows/ci.yml/badge.svg)](https://github.com/mounimb/git-shift/actions/workflows/ci.yml)
[![PyPI](https://img.shields.io/pypi/v/git-shift)](https://pypi.org/project/git-shift/)

**Folder-based git identities, with a guard against committing as the wrong account.**

If you use several GitHub accounts (personal, work, open source) on one machine, git's
single global identity means commits end up credited to the wrong account, and you only
notice after pushing. `git-shift` binds each folder to an identity (name, email, SSH key,
GitHub login), explains which identity applies anywhere, and blocks commits and pushes
made with the wrong one.

I built it because commits on my personal repositories kept landing on my other GitHub
account.

## Who it is for

Developers with more than one git identity on the same computer, on Windows, macOS or
Linux, from the terminal or any GUI client (VS Code, GitHub Desktop, IDEs).

## Install

```bash
pipx install git-shift      # or: pip install git-shift
```

Requires Python 3.10+ and git 2.13+. `gh` (GitHub CLI) is optional.

## Quick start

```bash
git shift add perso --email me@example.com --name "Jane Doe" \
    --folder ~/code/perso --ssh-key ~/.ssh/id_perso --gh jane
git shift add work --email jane@company.com --name "Jane Doe" --folder ~/code/work
git shift guard install

cd ~/code/perso/my-project
git shift status
```

```
repo:     /home/jane/code/perso/my-project
expected: "perso" <me@example.com>  (matched by folder /home/jane/code/perso/)
email:    me@example.com  (from identity "perso")
ssh:      ssh -i /home/jane/.ssh/id_perso -o IdentitiesOnly=yes
gh:       jane (active)
```

## Commands

| Command | What it does |
|---|---|
| `git shift add <name> --email E --name N --folder DIR [--ssh-key K] [--gh LOGIN]` | Create an identity bound to a folder |
| `git shift list` | List identities |
| `git shift status` | Which identity applies here, and why |
| `git shift use <name> --here` | Pin an identity on the current repo |
| `git shift remove <name>` | Delete an identity |
| `git shift doctor` | Check the setup, with a fix for each problem |
| `git shift guard install` / `uninstall` | Block commits and pushes made with the wrong identity |

Use `git shift -h` for help (`git shift --help` makes git look for a man page).

## How it works

- Each identity is a plain git config file in `~/.config/git-shift/identities/`.
- `~/.config/git-shift/rules.gitconfig` maps folders to identities with git's own
  conditional includes (`includeIf "gitdir/i:..."`). Your global config gets a single
  `include` line pointing to it. Because this is native git configuration, it works in
  every git client, even when `git-shift` is not running.
- The guard sets a global `core.hooksPath`. Its `pre-commit` hook checks the author
  email; its `pre-push` hook checks every commit about to be pushed. Repositories' own
  hooks keep running: the guard chains to them.
- A repository outside every identity folder is still protected when its `origin` is
  `github.com/<login>/...` and `<login>` belongs to one of your identities.
- `git-shift` changes git config only through `git config`, and backs up your global
  config before its first change.

## Limitations

- Repositories that set their own `core.hooksPath` (husky, pre-commit) bypass the
  global guard. `git shift doctor` tells you when that is the case.
- The guard does not chain `reference-transaction` and `post-index-change` hooks.
- The origin-owner check only recognises `github.com` remotes. SSH host aliases such as
  `git@github-work:owner/repo` are not checked yet (see the roadmap).
- Bypass the guard once with `git commit --no-verify` or `GIT_SHIFT_SKIP=1`.
